\begin{frame}{Fase 1: normalizar la entrada}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item \textbf{Hipótesis}: con píxeles en $[0, 255]$ los logits son enormes y la softmax se satura
        \item \textbf{Evidencia}: con píxeles crudos la pérdida de train no baja de $\sim$\res{baseline-train-loss}
        \item \textbf{Resultado}: con \texttt{Rescaling(1/255)} es dos órdenes de magnitud menor, pero sigue por encima de $\ln K$ hasta que baja la lr
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{\texttt{Rescaling(1/255)} dentro del modelo: +\res{gain-normalisation}~pp de mean recall.}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{figure}
        \centering
        \safegraphics[width=\textwidth,height=0.62\textheight,keepaspectratio]{figures/normalisation.png}{Pérdida con y sin escalado}
        \caption{\footnotesize \figcaptionshort{normalisation}}
      \end{figure}
    \end{column}
  \end{columns}
  \speakernote{1}{Lo razonable sería $\ln 13 \approx$ \res{chance-loss}; en la primera época con Rescaling aún es \res{norm-loss-first}. La caída final de la curva naranja es ReduceLROnPlateau actuando. Estandarizar por canal (EXP-004) empeora (\res{gain-standardise}~pp; pérdida inicial \res{std-loss-first}): el paso de Adam en un logit crece con $\sum_i |x_i|$. Segunda decisión de la fase: callbacks con paciencias \res{phase1-es-patience}/\res{phase1-rlr-patience} sobre el mean recall de validación (frente a pérdida o accuracy: \res{monitor-loss-mean-recall} y \res{monitor-acc-mean-recall}, una semilla); la oscilación de la accuracy de validación baja de \res{norm-valacc-std-last10} a \res{phase1-valacc-std-last10}. Cómo actúan se ve en el modelo final.}
\end{frame}

\begin{frame}{Fase 2: reducir la resolución de entrada}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item \textbf{Hipótesis}: a \res{res224-res}~$\times$~\res{res224-res} hay demasiados parámetros: sobreajuste
        \item \textbf{Evidencia}: al subir la resolución cae también train: no es sobreajuste sino optimización
        \item \textbf{Resultado}: máximo en \res{phase2-res}~$\times$~\res{phase2-res}, también con la mejor lr de cada tamaño (rombos)
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{Entrada de \res{phase2-res}~$\times$~\res{phase2-res}: +\res{gain-resolution}~pp, el mayor salto.}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{figure}

        \caption{\footnotesize \figcaptionshort{resolution_sweep}}

    \end{column}
  \end{columns}
  \speakernote{1}{Todas con una capa oculta de \res{phase2-width} unidades (ReLU, He) y reducción por promedio de área (sin aliasing). La causa de la caída: el paso de Adam en una preactivación crece con el número de entradas y mata ReLU; se mide en la fase 4 (\res{phase2-inactive}\,\% de inactivas a \res{phase2-res}, \res{res224-inactive}\,\% a \res{res224-res}). A \res{res224-res} la red pasa las primeras épocas en la distribución a priori. \res{res24-res}~$\times$~\res{res24-res} (\res{res24-mean-recall}) queda dentro del ruido.}

\begin{frame}{Fase 3: más capacidad no generaliza más}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item \textbf{Hipótesis}: más capacidad, mejor validación
        \item \textbf{Evidencia}: train sube hasta \res{big3layer-train-acc}; validación se queda en $\sim$\res{phase3-val-acc}
        \item \textbf{Resultado}: más allá de \res{phase2-params} parámetros la validación ya no mejora
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{\res{phase3-hidden} (\res{phase3-params} parámetros): igual que la red mayor, con menos brecha.}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{figure}

        \caption{\footnotesize \figcaptionshort{capacity}}

  \end{columns}
  \speakernote{1}{En mean recall, \res{phase3-hidden} da \res{phase3-mean-recall}, lo mismo que la red mayor (\res{big3layer-hidden}) con muchos menos parámetros, y más que una sola capa de 512 (\res{onelayer512-mean-recall}); una semilla, dentro del ruido. En todas las arquitecturas, entre el \res{arch-inactive-min} y el \res{arch-inactive-max}\,\% de las ReLU de la primera capa están muertas: de ahí la fase 4.}
\end{frame}

\begin{frame}{Fase 4: LeakyReLU contra las ReLU muertas}
  \begin{figure}
    \centering
    \safegraphics[width=\textwidth,height=0.5\textheight,keepaspectratio]{figures/activation.png}{Unidades inactivas y activaciones}
    \caption{\footnotesize \figcaptionshort{activation}}
  \end{figure}
  \vspace{-0.3cm}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{itemize}\small
        \item \textbf{Evidencia} (izq.): las ReLU inactivas crecen con la resolución y con la lr
        \item \textbf{Resultado} (dcha.): todas las alternativas las eliminan; LeakyReLU 0,3 es la mejor en validación
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \decisionblock{LeakyReLU ($\alpha$ = \res{phase4-leaky-alpha}) con He: \res{phase4-mean-recall} ($\pm$ \res{phase4-mean-recall-std}).}
    \end{column}
  \end{columns}
  \speakernote{1,5}{ReLU tiene derivada nula para $z<0$: una unidad negativa con todas las imágenes no recibe gradiente y no se recupera. LeakyReLU, $\alpha z$ para $z<0$, siempre tiene gradiente. Con la mejor lr de cada tamaño (rombos) la fracción de inactivas es parecida: lo que cuenta es el tamaño del paso. ReLU ya usaba He, así que solo cambia la activación. LeakyReLU 0,1 queda por debajo de ReLU (\res{leaky01-mean-recall}): quitar las muertas no basta, y +\res{gain-activation}~pp es del orden del ruido. ELU, GELU y SELU memorizan (train hasta \res{gelu-train-mean-recall}) sin mejorar validación; tanh, la peor.}
\end{frame}
