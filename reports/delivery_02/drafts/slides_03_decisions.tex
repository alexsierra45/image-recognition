        \caption{\footnotesize \figcaptionshort{l2_curve}}

  \end{columns}
  \begin{itemize}
    \item Al regularizar más, train y validación bajan juntas: ningún valor mejora la validación
  \end{itemize}
  \decisionblock{Sin Dropout ni L2 (y, sobre BN + rotaciones, tampoco ayudan).}
  \speakernote{1,25}{Dropout entrena muchas subredes que comparten pesos (Srivastava et al.); L2 encoge los pesos. Con $p$~$=$~\res{dropout05-dropout}, mean recall \res{dropout05-mean-recall} y \res{class-min-name} a recall \res{dropout05-recall-helipad}. Es el lado del subajuste de la curva: el límite parece el sesgo del modelo, no la varianza. Delante de BatchNorm, L2 solo cambia la lr efectiva (van Laarhoven).}
\end{frame}

\begin{frame}{BatchNorm: optimiza más que regulariza}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.4\textwidth}
      \begin{itemize}
        \item Con la misma lr empata, y el train sube: la brecha crece
        \item Antes de la activación y lr $\times$3: +\res{gain-bn}~pp
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{BatchNorm antes de la activación, lr \res{phase1-lr}. Con augmentation, quitarlo cuesta \res{loss-no-bn}~pp.}
    \end{column}
    \begin{column}{0.58\textwidth}
      \begin{figure}
        \centering
        \safegraphics[width=\textwidth,height=0.62\textheight,keepaspectratio]{figures/batchnorm.png}{BatchNorm}
        \caption{\footnotesize \figcaptionshort{batchnorm}}
      \end{figure}
    \end{column}
  \end{columns}
  \speakernote{1}{BatchNorm (Ioffe y Szegedy) hace la salida independiente de la escala de los pesos y permite una lr mayor. Solo aporta del orden del ruido; su peso aparece al combinarlo con la augmentation.}
\end{frame}

\begin{frame}{Data augmentation: rotaciones de 90$^\circ$}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.4\textwidth}
      \begin{itemize}
        \item En imágenes aéreas no hay ``arriba'': rotar conserva la etiqueta
        \item Rotaciones, volteos y D4: train baja y validación sube, la brecha se acorta
        \item Brillo y contraste: empeoran
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{Rotaciones de 90$^\circ$: +\res{gain-rot90}~pp.}
    \end{column}
    \begin{column}{0.58\textwidth}
      \begin{figure}
        \centering
        \safegraphics[width=\textwidth,height=0.62\textheight,keepaspectratio]{figures/augmentation.png}{Data augmentation}
        \caption{\footnotesize \figcaptionshort{augmentation}}
      \end{figure}
    \end{column}
  \end{columns}
  \speakernote{1}{La única técnica que reduce la brecha (a \res{phase2-gap-acc}~pp) subiendo la validación: codifica una invarianza que la ffNN no tiene. Volteos y D4 empatan dentro del ruido. Con augmentation, la mejor época llega más tarde.}
\end{frame}

  \end{itemize}
  \decisionblock{Se mantiene \res{final-hidden} sin Dropout ni L2.}
  \speakernote{1,25}{Dropout y L2 se habían juzgado sobre una red elegida para no sobreajustar; aquí se varía a la vez la anchura (hasta \res{w1024-params} parámetros). La interacción existe pero no basta: la mejor celda regularizada (512 con L2, \res{w512-l2-mean-recall}) empata con el modelo final.}
\end{frame}
