\begin{frame}{De \res{phase2-mean-recall} a \res{best-mean-recall}: la learning rate bajaba demasiado pronto}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item Con rotaciones, la validación oscila mucho entre épocas
        \item Con paciencia \res{final-rlr-patience}, la lr no baja hasta la época \res{stop-first-lr-drop}, y esa bajada da el salto
        \item Después solo mejora train

      \decisionblock{Paciencia de ReduceLROnPlateau \res{phase2-rlr-patience} $\rightarrow$ \res{final-rlr-patience}: +\res{gain-rlr8}~pp. La de EarlyStopping no influye (+\res{gain-es25}~pp).}
