\begin{frame}{Normalizar la entrada}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item \textbf{Hipótesis}: con píxeles en $[0, 255]$ los logits son enormes y la softmax se satura
        \item \textbf{Evidencia}: con píxeles crudos la pérdida de train no baja de $\sim$\res{baseline-train-loss}
        \item \textbf{Resultado}: con \texttt{Rescaling(1/255)} es dos órdenes de magnitud menor y baja al reducir la lr
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{\texttt{Rescaling(1/255)} dentro del modelo: +\res{gain-normalisation}~pp de mean recall.}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{figure}
        \centering
        \safegraphics[width=\textwidth,height=0.62\textheight,keepaspectratio]{figures/normalisation.png}{Pérdida con y sin escalado}
        \caption{\footnotesize \figcaptionshort{normalisation}}
      \end{figure}
    \end{column}
  \end{columns}
  \speakernote{1}{Lo razonable sería $\ln 13 \approx$ \res{chance-loss}. Al estar dentro del modelo, train, validación y test reciben el mismo preprocesado. La caída final de la curva naranja es ReduceLROnPlateau actuando.}
\end{frame}

\begin{frame}{Cuándo parar: lo decide la validación}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item Hasta la época \res{stop-last-joint-epoch}, train y validación suben juntas
        \item Al estancarse la validación, la lr baja $\times$0,1: último salto, mejor época (\res{stop-best-epoch})
        \item Después solo mejora train (\res{stop-train-acc-best} $\rightarrow$ \res{stop-train-acc-end}); validación, plana: memoriza
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{EarlyStopping (paciencia \res{final-es-patience}) sobre el mean recall: para en la época \res{stop-epoch} y recupera la \res{stop-best-epoch}.}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{figure}
        \centering
        \safegraphics[width=\textwidth,height=0.62\textheight,keepaspectratio]{figures/stopping.png}{Punto de parada}
        \caption{\footnotesize \figcaptionshort{stopping}}
      \end{figure}
    \end{column}
  \end{columns}
  \speakernote{1}{La brecha train-val pasa de \res{stop-gap-best}~pp en la mejor época a \res{stop-gap-end}~pp al parar. Las dos bajadas de lr posteriores ya no cambian nada. En el baseline los callbacks no podían actuar (paciencias \res{baseline-es-patience} y \res{baseline-rlr-patience} con \res{baseline-epochs-cap} épocas). Se vigila el mean recall porque es la métrica de decisión; vigilar la pérdida o la accuracy dio algo menos (una semilla). Las semillas 2 y 3 siguen el mismo patrón.}
\end{frame}

\begin{frame}{Arquitectura: más capacidad no generaliza más}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item \textbf{Hipótesis}: más capacidad, mejor validación
        \item \textbf{Evidencia}: train sube hasta \res{big3layer-train-acc}; validación se queda en $\sim$\res{phase3-val-acc}
        \item \textbf{Resultado}: por encima de $\sim 2\cdot 10^5$ parámetros la validación ya no mejora
      \end{itemize}
