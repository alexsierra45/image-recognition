\begin{frame}{Normalizar y reducir la resolución}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.4\textwidth}
      \begin{itemize}
        \item \textbf{Hipótesis}: con tantas entradas, sobra varianza
        \item \textbf{Evidencia}: a \res{res224-res} píxeles también cae train: casi todas las ReLU mueren
        \item \textbf{Resultado}: \texttt{Rescaling} +\res{gain-normalisation}~pp; \res{phase2-res}~$\times$~\res{phase2-res}, +\res{gain-resolution}~pp
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{Entrada de \res{phase2-res}~$\times$~\res{phase2-res}: el mayor salto del plan.}
    \end{column}
    \begin{column}{0.55\textwidth}

\begin{frame}{La regularización clásica no ayuda}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.4\textwidth}
      \begin{itemize}
        \item \textbf{Hipótesis}: la brecha train-val es sobreajuste
        \item \textbf{Evidencia}: con dropout o L2, train y validación bajan juntas
        \item \textbf{Resultado}: BatchNorm permite subir la lr a \res{phase6-lr}: +\res{gain-bn}~pp
      \end{itemize}
      \vspace{0.2cm}
      \decisionblock{BatchNorm, sin dropout ni L2: el límite es el sesgo del modelo.}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{figure}
        \centering
        \safegraphics[width=\textwidth]{figures/regularisation.png}{Curva de validación del dropout}
        \caption{\footnotesize \figcaption{regularisation}}
      \end{figure}
    \end{column}
  \end{columns}
  \speakernote{1}{Ruido entre semillas de \res{seed-noise-pp}~pp: por eso se decide con tres semillas. Batch \res{phase5-batch}: épocas \res{speedup-batch256} veces más rápidas.}
\end{frame}
